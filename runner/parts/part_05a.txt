    spec = make_group_spec(stats)
    groups = group_masks(stats, spec)
    rules, history = fit_booster("UC", p[:40], y[:40], p[40:], y[40:], {k: v[:40] for k, v in groups.items()}, {k: v[40:] for k, v in groups.items()}, tree, max_rounds=3, patience=2)
    return {
        "tree_nodes": tree.num_nodes,
        "max_decomposition_error": max(compute_utility_state(p, y, tree, r).decomposition_error for r in RANKS),
        "interval_score": exact.score,
        "projection_error": float(np.max(np.abs(projected.sum(axis=1) - 1))),
        "booster_rules": len(rules),
        "history_rows": len(history),
    }
