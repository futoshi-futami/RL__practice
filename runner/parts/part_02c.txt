@dataclass
class GroupSpec:
    brightness_median: float
    contrast_median: float


def make_group_spec(train_stats: Mapping[str, np.ndarray]) -> GroupSpec:
    return GroupSpec(
        brightness_median=float(np.median(train_stats["brightness"])),
        contrast_median=float(np.median(train_stats["contrast"])),
    )


def group_masks(stats: Mapping[str, np.ndarray], spec: GroupSpec) -> Dict[str, np.ndarray]:
    brightness = np.asarray(stats["brightness"], dtype=np.float64)
    contrast = np.asarray(stats["contrast"], dtype=np.float64)
    aspect = np.asarray(stats["aspect"], dtype=np.float64)
    return {
        "all": np.ones(len(brightness), dtype=bool),
        "dark": brightness <= spec.brightness_median,
        "bright": brightness > spec.brightness_median,
        "low_contrast": contrast <= spec.contrast_median,
        "high_contrast": contrast > spec.contrast_median,
        "portrait": aspect < 1.0,
        "landscape": aspect >= 1.0,
    }

def candidate_sort_key(candidate: Candidate) -> Tuple[float, int, str, int, float, float]:
    utility_order = {r: i for i, r in enumerate(RANKS)}
    return (
        -candidate.score,
        utility_order[candidate.utility_rank],
        candidate.group,
        -1 if candidate.node_index is None else candidate.node_index,
        candidate.lower,
        candidate.upper,
    )
